@dataclass
class Account:
    id: int
    email: str
    password_hash: str
    status: str = "active"

def sign_in(email, password, accounts, sessions, hasher):
    account = accounts.find_by_email(email)
    if account is None or account.status != "active":
        raise InvalidCredentials()
    if not hasher.verify(password, account.password_hash):
        raise InvalidCredentials()
    return sessions.create(account.id)
